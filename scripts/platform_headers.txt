# Platform layout for scripts/check_platform_headers.py.
#
# A platform is a directory with include/ under src/kernel/platform/ and
# src/platform/ (both trees use the same names). Every such directory is a
# platform of its own, unless it is listed in [groups].

# platform: the platform header directories its builds use together, as
# zephyr/CMakeLists.txt puts them on the include path
[groups]
tigerlake: tigerlake intel/cavs
meteorlake: meteorlake intel/ace
lunarlake: lunarlake intel/ace
ace30: ace30 intel/ace
novalake: novalake intel/ace
rembrandt: amd/rembrandt amd/common
acp_7_0: amd/acp_7_0 amd/common
acp_7_x: amd/acp_7_x amd/common

# platform name pattern: the vendor driver directory its builds compile
[vendor]
imx*: src/kernel/drivers/imx/
rembrandt: src/kernel/drivers/amd/
acp_7_*: src/kernel/drivers/amd/
mtk: src/kernel/drivers/mediatek/

# platforms of the host builds (host library, testbench, cmocka): they see
# the host-only code and not the Zephyr glue
[host]
library

# platforms of the native_sim builds: they also see the ztest unit tests
[posix]
posix
