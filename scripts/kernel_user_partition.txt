# Kernel / user-space partition of the SOF sources, used by
# scripts/check_kernel_includes.py.
#
# Kernel code runs in Zephyr supervisor mode: hardware and MMIO access,
# interrupts, thread and memory domain creation, cache/MMU, power and
# clock management, DMA and DAI driver internals, IDC, LLEXT loading and
# system call verification. User-space code is everything else and must
# only use plain C, system-call backed Zephyr APIs and SOF system calls,
# whether or not a given configuration actually runs it in user mode.
#
# Patterns are fnmatch globs relative to the SOF top directory, except
# [zephyr-kernel-headers] which is relative to ZEPHYR_BASE. "*" also
# matches "/".

# Sources that must be user-space safe: everything in src/ that is not
# kernel code. Kernel code is in src/kernel/ and the zephyr/ directory.
[user-sources]
src/*

# Kernel sources inside src/: src/kernel/, and the legacy drivers and LL
# scheduling domains, which go with the legacy drivers.
[not-user-sources]
src/kernel/*
src/audio/dai-legacy.c
src/audio/host-legacy.c
src/schedule/dma_multi_chan_domain.c
src/schedule/dma_single_chan_domain.c
src/schedule/zephyr_dma_domain.c

# Headers that user-space code may include, checked with --sources: they
# must not include headers from src/kernel/include. The kernel sources
# above and the [kernel-headers] below are excluded.
[user-headers]
src/*
zephyr/include/*

# SOF headers that only kernel code may include.
# Not listed on purpose: rtos/init.h, which only provides the
# SOF_MODULE_INIT() registration hook used by every module, and
# sof/lib/mailbox.h: the mailbox windows are mapped into the user-space
# LL and IPC threads, and their layout is in the user-space platform
# headers in src/platform.
[kernel-headers]
src/kernel/include/*

src/include/sof/lib/cpu-clk-manager.h
src/include/sof/lib/perf_cnt.h
src/include/sof/lib/watchdog.h

zephyr/include/rtos/interrupt.h
zephyr/include/rtos/interrupt-legacy.h
zephyr/include/rtos/timer.h

zephyr/include/sof/lib/pm_runtime.h

# Zephyr headers that only kernel code may include: kernel internals and
# driver APIs that are not system calls.
# Not listed on purpose: zephyr/drivers/dma.h. Its types and constants are
# the arguments of the sof_dma_*() system calls in sof/lib/sof_dma.h, so
# user-space code needs them. User-space code must not call its dma_*()
# driver functions directly, which this include check cannot catch.
[zephyr-kernel-headers]
kernel/include/*
include/zephyr/pm/device_runtime.h
